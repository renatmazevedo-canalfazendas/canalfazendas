  <script>
    const MAX_MIDIAS = 10;
    const CAPA_PADRAO = 'https://images.unsplash.com/photo-1464226184884-fa280b87c399?auto=format&fit=crop&w=1200&q=80';

    function ehVideo(url) {
      return /\.(mp4|webm|ogg|mov)(\?|$)/i.test(url || '');
    }

    async function carregarDestaquesYoutube() {
      const grid = document.getElementById('grid-destaques-yt');
      let itens = [];

      try {
        const res = await fetch('destaques-youtube.json?t=' + Date.now());
        if (res.ok) {
          const data = await res.json();
          itens = Array.isArray(data.itens) ? data.itens : [];
        }
      } catch (e) {}

      if (!itens.length) {
        itens = [{
          youtubeUrl: 'https://www.youtube.com/@canalfazendas',
          capa: CAPA_PADRAO,
          textoDestaque: 'Essa fazenda tem água por todo lado! 571 ha em Jataí-GO'
        }];
      }

      grid.innerHTML = itens.map((item) => {
        const url = item.youtubeUrl && item.youtubeUrl.trim() ? item.youtubeUrl.trim() : 'https://www.youtube.com/@canalfazendas';
        const capa = item.capa && item.capa.trim() ? item.capa.trim() : CAPA_PADRAO;
        const texto = item.textoDestaque || '';

        return `
          <a
            href="${url}"
            target="_blank"
            rel="noopener noreferrer"
            class="group relative block h-64 sm:h-72 rounded-2xl overflow-hidden border border-white/15 bg-slate-900 shadow-xl"
          >
            <img
              src="${capa}"
              alt="${texto}"
              class="w-full h-full object-cover transition-transform duration-300 group-hover:scale-105"
            />
            <div class="absolute inset-0 bg-gradient-to-t from-black/90 via-black/35 to-black/10"></div>
            <div class="absolute bottom-4 left-4 right-4">
              <p class="text-white font-black text-lg sm:text-xl leading-snug whitespace-pre-line group-hover:text-teal-300 transition-colors">
                ${texto}
              </p>
              <span class="inline-block mt-2 text-xs font-extrabold text-red-400">
                ▶ Assistir no YouTube
              </span>
            </div>
          </a>
        `;
      }).join('');
    }

    async function carregarFazendas() {
      const container = document.getElementById('lista-fazendas');
      let fazendas = [];

      try {
        const res = await fetch('fazendas.json?t=' + Date.now());
        if (res.ok) {
          const data = await res.json();
          fazendas = Array.isArray(data.itens) ? data.itens : (Array.isArray(data) ? data : []);
        }
      } catch (e) {}

      if (!fazendas.length) {
        container.innerHTML = `
          <div class="bg-[#121b19] border border-white/10 rounded-2xl p-6 text-slate-300 text-sm">
            Nenhuma fazenda cadastrada ainda. Acesse <a href="/admin/" class="text-teal-400 underline font-bold">/admin</a> para adicionar.
          </div>
        `;
        return;
      }

      container.innerHTML = fazendas.map((f, fIdx) => {
        const midiasBrutas = Array.isArray(f.midias) ? f.midias.slice(0, MAX_MIDIAS) : [];
        const midias = midiasBrutas
          .map(m => ({ url: (m.arquivo || m.url || '').trim(), legenda: m.legenda || '' }))
          .filter(m => m.url);

        const principal = midias[0] || { url: CAPA_PADRAO, legenda: '' };

        const galeriaHtml = midias.length > 0
          ? `<div class="grid grid-cols-2 sm:grid-cols-5 gap-2.5 mt-4">${midias.map((m, mIdx) => {
              const conteudo = ehVideo(m.url)
                ? `<video src="${m.url}" controls playsinline class="w-full h-24 object-cover bg-black"></video>`
                : `<img src="${m.url}" alt="${m.legenda || f.titulo}" class="w-full h-24 object-cover cursor-pointer" onclick="selecionarMidia(${fIdx}, ${mIdx})" />`;
              return `<div class="rounded-xl overflow-hidden border border-white/10 bg-slate-900">${conteudo}</div>`;
            }).join('')}</div>`
          : '';

        const zapMsg = encodeURIComponent(`Olá! Tenho interesse na fazenda: ${f.titulo}`);

        return `
          <div class="bg-[#121b19] border border-white/10 rounded-2xl overflow-hidden grid grid-cols-1 lg:grid-cols-12">
            <div id="midia-principal-${fIdx}" class="lg:col-span-6 min-h-[260px] bg-slate-900">
              ${ehVideo(principal.url)
                ? `<video src="${principal.url}" controls playsinline class="w-full h-full min-h-[260px] object-cover bg-black"></video>`
                : `<img src="${principal.url}" alt="${f.titulo}" class="w-full h-full min-h-[260px] object-cover" />`
              }
            </div>

            <div class="lg:col-span-6 p-6 flex flex-col justify-between gap-6">
              <div>
                <h3 class="text-xl sm:text-2xl font-black text-white mb-3">${f.titulo}</h3>
                <p class="text-slate-300 text-sm sm:text-base leading-relaxed whitespace-pre-line">${f.descricao || ''}</p>
                ${galeriaHtml}
              </div>

              <a
                href="https://wa.me/5564996463671?text=${zapMsg}"
                target="_blank"
                rel="noopener noreferrer"
                class="w-full sm:w-auto self-start bg-emerald-500 hover:bg-emerald-400 text-slate-950 font-extrabold text-sm px-6 py-3.5 rounded-xl transition"
              >
                Falar no WhatsApp
              </a>
            </div>
          </div>
        `;
      }).join('');

      window.__FAZENDAS_CACHE__ = fazendas;
    }

    function selecionarMidia(fIdx, mIdx) {
      const f = (window.__FAZENDAS_CACHE__ || [])[fIdx];
      if (!f || !Array.isArray(f.midias)) return;
      const m = f.midias[mIdx];
      const url = m && (m.arquivo || m.url);
      const box = document.getElementById(`midia-principal-${fIdx}`);
      if (!url || !box) return;
      box.innerHTML = ehVideo(url)
        ? `<video src="${url}" controls playsinline class="w-full h-full min-h-[260px] object-cover bg-black"></video>`
        : `<img src="${url}" alt="${f.titulo}" class="w-full h-full min-h-[260px] object-cover" />`;
    }

    carregarDestaquesYoutube();
    carregarFazendas();
  </script>
